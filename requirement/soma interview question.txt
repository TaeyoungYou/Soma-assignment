  DateTime? end;

  List<BandLocations> activeBands = [];

  int get sampleCount {
    var n = 0;
    for (final band in dat0) {
      for (final chan in band) {
        n += chan.length;
      }
    }
    for (final band in dat1) {
      n += band.length;
    }
    for (final band in dat2) {
      for (final analyte in band) {
        n += analyte.length;
      }
    }
